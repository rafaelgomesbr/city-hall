def grava(pasta, nome, texto):
    with open(os.path.join(S, pasta, nome), "w", encoding="utf-8", newline="\n") as f:
        f.write(texto)


def troca(t, velho, novo):
    if t.count(velho) != 1:
        raise SystemExit(f"esperava 1x: {velho!r} (achei {t.count(velho)})")
    return t.replace(velho, novo)

def e08(grupos, comentario):
    linhas = []
    for privilegio, colunas, redefine in grupos:
        linhas.append(
            "    {\n"
            f'      tabela: "usuarios", papel: "dc360_app", privilegio: "{privilegio}",\n'
            f"      colunas: {list(colunas)!r},\n".replace("'", '"') +
            '      migracao: "061", etapa: ETAPA, motivo: MOTIVO,' + (" redefine: true," if redefine else "") + "\n"
            "    },\n"
        )
    return (
        f"// {comentario}\n"
        'import type { ContribuicaoDoCatalogo } from "./tipos.js";\n\n'
        'const ETAPA = "V2-F03-E08";\n'
        'const MOTIVO = "Sabotagem da correcao 1 da 0G: a E08 encolhe o privilegio de usuarios (passo 3 da P-04).";\n\n'
        "export const CATALOGO_F03_E08: ContribuicaoDoCatalogo = {\n"
        "  funcoes: [],\n  excecoes: [],\n  colunasComUpdate: [],\n"
        "  privilegiosDeColuna: [\n" + "".join(linhas) + "  ],\n};\n"
    )


def sql061(update, comentario):
    return (
        f"-- {comentario}\n"
        "REVOKE SELECT, INSERT, UPDATE ON usuarios FROM dc360_app;\n"
        f"GRANT SELECT ({', '.join(SELECT)}) ON usuarios TO dc360_app;\n"
        f"GRANT INSERT ({', '.join(INSERT)}) ON usuarios TO dc360_app;\n"
        f"GRANT UPDATE ({', '.join(update)}) ON usuarios TO dc360_app;\n"
    )
